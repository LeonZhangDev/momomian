    errors = []
    if set(args) - {"date", "people"}:

对照参考实现与逐步解析
def unique_requests(requests):
    seen = set()
    result = []
    for request in requests:
        if request["id"] not in seen:
            seen.add(request["id"])
            result.append(request)
    return result

怎么推出来

  1. seen 保存已经出现的 id，result 保存最终完整请求，两者职责不同。
  2. 只有 id 未出现时才同时加入 seen 和 result；遍历一次即可。
  3. 重复 ID 的字段不同也保留第一条，不能用后一次值覆盖前一次。

走一遍例子：a(value=1) 入列 → 第二个 a(value=2) 跳过 → b(value=3) 入列，结果仍为第一条 a 和 b。

用这个反例检查：dict[id]=request 会覆盖成最后一次请求，违背“保留首次”的约定。

时间复杂度：平均 O(n)；空间复杂度：O(u)。

n 为请求数，u 为不同 ID 数；集合查询按平均常数时间估算。

def chunk_tokens(tokens, size, overlap):
    if size <= 0 or overlap < 0 or overlap >= size:
        return []
    chunks = []
    start = 0
    while start < len(tokens):

def fuse_ranks(rankings, k, limit):
    if limit <= 0:
        return []
    scores = {}
    for ranking in rankings:
        seen = set()
        for rank, doc in enumerate(ranking, 1):
            if doc in seen:
                continue
            seen.add(doc)
            scores[doc] = scores.get(doc, 0) + 1 / (k + rank)
    ranked = sorted(scores, key=lambda doc: (-scores[doc], doc))
    return ranked[:limit]

def simulate_lru(capacity, operations):
    cache, order, results = {}, [], []
    for operation in operations:
        action, key = operation[:2]
        if action == "get":
            results.append(cache.get(key, -1))
            if key not in cache:
                continue
        else:
            cache[key] = operation[2]
        if key in order:
            order.remove(key)
        order.append(key)
        if len(order) > capacity:
            del cache[order.pop(0)]
    return results

def dependency_levels(nodes, edges):
    indegree = {node: 0 for node in nodes}
    graph = {node: [] for node in nodes}
    seen = set()
    for a, b in edges:
        edge = (a, b)
        if edge not in seen:
            seen.add(edge)
            graph[a].append(b)
            indegree[b] += 1
    remaining = list(nodes)
    levels = []
    while remaining:
        level = [node for node in remaining if indegree[node] == 0]
        if not level:
            return []
        levels.append(level)
        active = set(level)
        remaining = [node for node in remaining if node not in active]
        for node in level:
            for following in graph[node]:
                indegree[following] -= 1
    return levels

  3. 空输入返回题目约定的 None；不能取索引 -1。

走一遍例子：n=20 时 ceil19=19，取排序后第19项；[1,2,3,100] 的P95按本题为100。

用这个反例检查：把 0.95*n 直接截断再当零基索引，在边界样本数会偏移。

时间复杂度：O(n log n)；空间复杂度：O(n)。

参考实现排序会创建有序列表；小样本分位数较粗，不能凭几个点推断长期尾延迟。
