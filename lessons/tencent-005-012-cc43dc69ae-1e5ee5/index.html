query = "如何用 Python 做人工智能"

results = collection.query(
    query_texts=[query],
    n_results=3, # 返回最相似的 3 条
    include=["documents", "distances", "metadatas"]
)

print(f"\n查询：{query}")
print("-" * 50)
for i, (doc, dist) in enumerate(zip(
    results["documents"][0],
    results["distances"][0]
)):
    similarity = 1 - dist   # 余弦距离转相似度
 print(f"第 {i+1} 名（相似度 {similarity:.4f}）：")
 print(f"  {doc}")
 print()
