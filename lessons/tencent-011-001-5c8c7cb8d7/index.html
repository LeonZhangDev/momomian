brain.py 文件代码

# brain.py
from openai import OpenAI
import os


DEEPSEEK_API_KEY = "sk-xxxxxxx" # 这里设置你申请的 key
DEEPSEEK_API_URL = "https://api.deepseek.com/v1"

client = OpenAI(
    api_key = DEEPSEEK_API_KEY,
    base_url = DEEPSEEK_API_URL)

class AgentBrain:
 """Agent 的大脑，负责思考与决策"""

 def __init__(self, model="deepseek-v4-flash"):
        self.model = model

 def think(self, prompt):
 """核心思考函数：接收提示，返回模型的思考结果"""
 try:
 # 使用客户端调用 Chat Completions API（v1.x 版本写法）
            response = client.chat.completions.create(
                model=self.model,
                messages=[{"role": "user", "content": prompt}],
                temperature=0.5, # 控制创造性，越低越专注
                max_tokens=500 # 控制回复长度
 )
 # 提取模型返回的文本内容（v1.x 版本属性路径变更）
            reasoning = response.choices[0].message.content
 return reasoning.strip()
 except Exception as e:
 return f"思考过程出错: {e}"

# 简单测试一下大脑是否工作
if __name__ == "__main__":
    brain = AgentBrain()
    test_prompt = "你好，请简单介绍一下你自己。"
 print("测试提问：", test_prompt)
 print("大脑回复：", brain.think(test_prompt))

代码解析：

  1. 我们创建了一个 AgentBrain 类来封装与 LLM 的交互。
  1. think 方法是核心，它接收一个字符串 prompt（提示词），将其发送给 GPT API。

  1. 更正：先确认目标模型及模式是否支持 temperature。支持时再通过重复任务评测取值，不能仅凭 0.5 推断稳定性与创造性的平衡。

    查看解释与参考资料

    0.5 不是规划任务通用最佳值。DeepSeek 的 thinking 模式不支持 temperature 调节；使用非思考模式时也需以任务评测选择。

    DeepSeek 思考模式 ↗

    temperature 参数很重要：设置为 0.5 能让 Agent 在创造性和稳定性之间取得平衡，适合做规划任务。
  1. 最后返回模型生成的文本，这就是 Agent 思考的结果。
